        function tick(now) {
          const p = Math.min((now - start) / duration, 1);
          const eased = 1 - Math.pow(1 - p, 3);
          el.textContent = Math.round(target * eased);
          if (p < 1) requestAnimationFrame(tick);
        }
        requestAnimationFrame(tick);
        zoneIO.unobserve(el);
      }
    });
  }, { threshold: 0.5 });

  const zoneBike = document.getElementById('zoneBike');

  let zoneBikeAnimating = false;
  
  const zoneBikeIO = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting && !zoneBikeAnimating) {
        zoneBikeAnimating = true;
        animateZoneBike();
      }
    });
  }, { threshold: 0.3 });
  zoneBikeIO.observe(zonesSection);

  function animateZoneBike() {
    let start = null;
    const duration = 4000;
    const containerWidth = zonesSection.offsetWidth - 100; // Account for icon width
    
    function step(timestamp) {
      if (!start) start = timestamp;
      const progress = Math.min((timestamp - start) / duration, 1);
      const eased = progress < 0.5 
        ? 2 * progress * progress 
        : 1 - Math.pow(-2 * progress + 2, 2) / 2;
      
      // Curve the path slightly
      const x = eased * containerWidth;
      const y = Math.sin(eased * Math.PI) * -15;
      zoneBike.style.left = `${x}px`;
      zoneBike.style.transform = `translateY(${y}px)`;
      
      if (progress < 1) {
        requestAnimationFrame(step);
      } else {
        // Restart after pause
        setTimeout(() => {
          start = null;
          requestAnimationFrame(step);
        }, 1500);
      }
    }
    requestAnimationFrame(step);
  }
